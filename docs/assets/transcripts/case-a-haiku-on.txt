import math

# 주어진 정보
revenue_monthly_base = 30_000_000  # 월 3천만 원
growth_rate = 0.30  # 월 30% 성장
fixed_cost_monthly = 80_000_000  # 월 고정비 8천만 원
initial_cash = 600_000_000  # 현금 6억 원

# 월별 계산
months = []
cash_remaining = initial_cash

for month in range(1, 37):  # 36개월 계산
    revenue = revenue_monthly_base * ((1 + growth_rate) ** (month - 1))
    operating_loss = fixed_cost_monthly - revenue
    cash_remaining -= operating_loss
    
    months.append({
        'month': month,
        'revenue': revenue,
        'cost': fixed_cost_monthly,
        'profit': revenue - fixed_cost_monthly,
        'cash': cash_remaining
    })
    
    # 현금이 음수가 되는 순간 멈춤 (런웨이 끝)
    if cash_remaining < 0 and months[-2]['cash'] >= 0:
        runway_month = month
        break

# 흑자 전환 시점
breakeven_month = None
for m in months:
    if m['profit'] > 0:
        breakeven_month = m['month']
        break

print("=" * 70)
